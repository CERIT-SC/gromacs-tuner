1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
  External access via Ingress:
  https://{{ .Values.ingress.host }}/api
{{- else }}
  Internal access via Service:
  http://gromacs-tuner-api-svc.{{ .Release.Namespace }}.svc.cluster.local:8000

  Or use port-forward for external access:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app=gromacs-tuner-api" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:8000
  echo "Visit http://127.0.0.1:8080 to use your application"
{{- end }}
